cmake_minimum_required(VERSION 3.20)
project(common_g_diagnostic LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
get_filename_component(ROOT "${CMAKE_CURRENT_LIST_DIR}/../../../.." ABSOLUTE)
if(NOT CORE_BUILD)
  message(FATAL_ERROR "CORE_BUILD must name the frozen diagnostic simulator/cpp build")
endif()
add_executable(common_g_execution execution.cpp)
target_include_directories(common_g_execution PRIVATE "${ROOT}/cpp/include")
target_link_libraries(common_g_execution PRIVATE
  "${CORE_BUILD}/liboriginal_grasu_cycle.a" "${CORE_BUILD}/libspine_cycle_core.a")
target_compile_options(common_g_execution PRIVATE -Wall -Wextra -Wpedantic -Werror)
add_executable(sharded_g_execution sharded/execution.cpp)
target_include_directories(sharded_g_execution PRIVATE "${ROOT}/cpp/include")
target_link_libraries(sharded_g_execution PRIVATE
  "${CORE_BUILD}/liboriginal_grasu_cycle.a" "${CORE_BUILD}/libspine_cycle_core.a")
target_compile_options(sharded_g_execution PRIVATE -Wall -Wextra -Wpedantic -Werror)
add_executable(real_sharded_g_execution sharded/real_execution.cpp)
target_include_directories(real_sharded_g_execution PRIVATE "${ROOT}/cpp/include")
target_link_libraries(real_sharded_g_execution PRIVATE
  "${CORE_BUILD}/liboriginal_grasu_cycle.a" "${CORE_BUILD}/libspine_cycle_core.a")
target_compile_options(real_sharded_g_execution PRIVATE -Wall -Wextra -Wpedantic -Werror)
enable_testing()
add_executable(ordered_g_input_test sharded/ordered_input_test.cpp)
target_include_directories(ordered_g_input_test PRIVATE "${ROOT}/cpp/include")
target_compile_options(ordered_g_input_test PRIVATE -Wall -Wextra -Wpedantic -Werror)
add_test(NAME ordered_g_input_test COMMAND ordered_g_input_test)
add_executable(metadata_layout_g_execution sharded/metadata_layout/execution.cpp)
target_include_directories(metadata_layout_g_execution PRIVATE "${ROOT}/cpp/include")
target_link_libraries(metadata_layout_g_execution PRIVATE
  "${CORE_BUILD}/liboriginal_grasu_cycle.a" "${CORE_BUILD}/libspine_cycle_core.a")
target_compile_options(metadata_layout_g_execution PRIVATE -Wall -Wextra -Wpedantic -Werror)
add_executable(resident_metadata_g_execution sharded/metadata_layout/lifecycle/main.cpp)
target_include_directories(resident_metadata_g_execution PRIVATE "${ROOT}/cpp/include")
target_link_libraries(resident_metadata_g_execution PRIVATE
  "${CORE_BUILD}/liboriginal_grasu_cycle.a" "${CORE_BUILD}/libspine_cycle_core.a")
target_compile_options(resident_metadata_g_execution PRIVATE -Wall -Wextra -Wpedantic -Werror)
